\section{Metodologia}

\begin{frame}
	
	\frametitle{Metodologia experimental}
	\textbf{Modelos comparados:}
	\begin{itemize}
		\item \textbf{EfficientNet-B0}: Modelo principal da comparação, com 5,3 milhões de parâmetros.
		\item \textbf{MobileNetV2 }: Alternativa mais leve, com 3,5 milhões de parâmetros.
		\item \textbf{ResNet18}: Arquitetura de referência, com 11,7 milhões de parâmetros.
	\end{itemize}
	
	\textbf{Configuração experimental:}
	\begin{itemize}
		\item Transfer learning com pesos pré-treinados na ImageNet.
		\item Três épocas iniciais de warmup.
		\item Fine-tuning com early stopping.
		\item Mesmos conjuntos de treino, validação e teste para a comparação controlada.
		\item Macro-F1 como métrica principal.
	\end{itemize}
			
\end{frame}
